async function loadDataSourceEntry(
  key: string,
  source: LowCodePageDataSource,
  pageBlocks: LowCodePageBlock[]
) {
  // A data source can be shared by more than one node. Let every node own its
  // loadData request so node action hooks and node-specific relation rules run.
  const nodeAction = resolveLowCodeDataSourceNodeAction(pageBlocks, key);
  const loaderBlocks = pageBlocks.filter((block) => (
    (block.kind === 'form'
      ? block.sourceKey ?? block.submitSourceKey
      : 'sourceKey' in block
        ? block.sourceKey
        : undefined) === key &&
    resolveLowCodeNodeAction(block.kind, 'loadData', block)?.dataSourceLoader === true
  ));
  if (nodeAction && !loaderBlocks.some((block) => block.id === nodeAction.block.id)) {
    loaderBlocks.unshift(nodeAction.block);
  }

  if (loaderBlocks.length) {
    if (source.autoLoad === false) return '';
    const errors = await Promise.all(loaderBlocks.map(async (block) => {
      try {
        await executeScriptNodeAction({
          node: block.id,
          method: 'loadData',
        });
        return '';
      } catch (error) {
        return `${key}/${block.id}: ${error instanceof Error ? error.message : host.t('runtime.errors.loadDataSource')}`;
      }
    }));
    return errors.filter(Boolean).join('; ');
  }

  const version = beginSourceRequest(key);
  return invokeDataSource(key, source)
    .then(([resolvedKey, value]) => {
      if (!isCurrentSourceRequest(key, version)) return '';
      if (typeof value !== 'undefined') {
        runtime.setSource(resolvedKey, value, { resetGridBaseline: true });
      }
      return '';
    })
    .catch((error: unknown) => {
      if (!isCurrentSourceRequest(key, version)) return '';
      return `${key}: ${error instanceof Error ? error.message : host.t('runtime.errors.loadDataSource')}`;
    })
    .finally(() => finishSourceRequest(key, version));
}
